  <!-- Interactive Logic Script -->
  <script>
    document.addEventListener('DOMContentLoaded', function () {
      const btnCalculate = document.getElementById('btnCalculateSubmit');
      const resultSection = document.getElementById('resultSection');
      const scoreDisplay = document.getElementById('scoreDisplay');
      const feedbackAlert = document.getElementById('feedbackAlert');

      btnCalculate.addEventListener('click', function () {
        // Retrieve Form Values
        const elec = parseFloat(document.getElementById('electricity').value) || 0;
        const gas = parseFloat(document.getElementById('naturalGas').value) || 0;
        const miles = parseFloat(document.getElementById('carMileage').value) || 0;
        const flightsVal = parseFloat(document.getElementById('flights').value) || 0;
        const recyclingVal = document.getElementById('recycling').value;
        const dietVal = document.getElementById('diet').value;

        // Approximate CO2 calculations (Metric tons per year)
        let totalCO2 = 0;

        // Energy (Electricity & Gas)
        totalCO2 += (elec * 12 * 0.0005);
        totalCO2 += (gas * 12 * 0.0008);

        // Transportation (Weekly km * 52)
        totalCO2 += (miles * 52 * 0.00021);

        // Flights
        totalCO2 += (flightsVal * 0.6);

        // Diet adjustment
        if (dietVal === 'heavy') totalCO2 += 2.5;
        else if (dietVal === 'average') totalCO2 += 1.7;
        else if (dietVal === 'vegetarian') totalCO2 += 0.9;

        // Recycling credit
        if (recyclingVal === 'full') totalCO2 -= 0.4;
        else if (recyclingVal === 'paper') totalCO2 -= 0.2;

        if (totalCO2 < 0.5) totalCO2 = 0.5;

        // Update UI
        scoreDisplay.textContent = totalCO2.toFixed(2);

        // Feedback messaging based on footprint size
        if (totalCO2 <= 4.0) {
          scoreDisplay.className = "display-4 fw-extrabold text-success";
          feedbackAlert.className = "alert alert-success border-0 rounded-3 mb-3";
          feedbackAlert.innerHTML = `
            <strong><i class="fa-solid fa-circle-check me-2"></i>Excellent Job!</strong> 
            Your carbon footprint is lower than the global average. Keep up your eco-friendly habits and encourage others to reduce their emissions!
          `;
        } else if (totalCO2 <= 7.5) {
          scoreDisplay.className = "display-4 fw-extrabold text-warning";
          feedbackAlert.className = "alert alert-warning border-0 rounded-3 mb-3";
          feedbackAlert.innerHTML = `
            <strong><i class="fa-solid fa-triangle-exclamation me-2"></i>Moderate Footprint</strong> 
            You are near the average footprint. Consider switching to public transit, reducing energy usage, or increasing recycling to lower your score!
          `;
        } else {
          scoreDisplay.className = "display-4 fw-extrabold text-danger";
          feedbackAlert.className = "alert alert-danger border-0 rounded-3 mb-3";
          feedbackAlert.innerHTML = `
            <strong><i class="fa-solid fa-circle-exclamation me-2"></i>High Footprint</strong> 
            Your emissions are above global target levels. Try reducing vehicle travel, switching off unnecessary appliances, and incorporating plant-based meals into your routine.
          `;
        }

        // Reveal result container with smooth view
        resultSection.classList.remove('d-none');
        resultSection.scrollIntoView({ behavior: 'smooth' });
      });
    });
  </script>
